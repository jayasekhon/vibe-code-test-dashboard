function extractArray(obj){ console.log("FULL JSON:", obj); // direct array if (Array.isArray(obj)) return obj; // ✅ FTS / flow common pattern if (obj.data && obj.data.flows) return obj.data.flows; if (obj.data && obj.data.records) return obj.data.records; if (obj.flows && Array.isArray(obj.flows)) return obj.flows; // fallback: find first array for (let key in obj) { if (Array.isArray(obj[key])) { console.log("Using fallback key:", key); return obj[key]; } } console.log("No usable array found"); return []; }
